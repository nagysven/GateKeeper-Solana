async def main():
    gk = GatekeeperAsyncClient(api_key="gk_production_key", base_url="https://gk.ai-futures-bot.pro")
    verdict = await gk.evaluate_intent(
        token_in="So11111111111111111111111111111111111111112",    # SOL

        amount_in=100_000_000,                                    # 0.1 SOL
        max_slippage_bps=50,
    )
    if verdict.is_approved:
        print(f"✅ Approved in {verdict.execution_time_ms}ms! Clamped CU: {verdict.clamped_compute_units}")

        print(f"🛡️ HARD_ABORT: {verdict.rejection_reason} - 0 Gas Burned!")

asyncio.run(main())
